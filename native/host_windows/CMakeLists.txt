cmake_minimum_required(VERSION 3.14)
project(neutrino_host LANGUAGES CXX)
add_executable(neutrino_host WIN32 main.cpp ../../windows/runner/Runner.rc)
target_compile_features(neutrino_host PRIVATE cxx_std_17)
target_compile_definitions(neutrino_host PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(neutrino_host PRIVATE /W4 /utf-8 /EHsc)
target_include_directories(neutrino_host PRIVATE ../../windows/runner ../third_party)
target_link_libraries(neutrino_host PRIVATE shell32 user32 gdi32 advapi32 crypt32 ole32)
set_target_properties(neutrino_host PROPERTIES OUTPUT_NAME neutrino_proxy_host)
